Pengaturan Supabase Cloud

Hubungkan TugasKu ke database PostgreSQL Supabase gratis agar tugas kuliahmu tersimpan permanen di cloud dan bisa sinkron otomatis antara HP & laptop secara real-time.

Skrip SQL Tabel `tugas_kuliah` 
CREATE TABLE IF NOT EXISTS public.tugas_kuliah (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  subject TEXT NOT NULL,
  name TEXT NOT NULL,
  deadline TIMESTAMPTZ NOT NULL,
  submission TEXT NOT NULL,
  is_done BOOLEAN DEFAULT FALSE,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Buka akses Row Level Security untuk publik
ALTER TABLE public.tugas_kuliah ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Public tugas access" ON public.tugas_kuliah 
FOR ALL USING (true) WITH CHECK (true);

-- Aktifkan Real-Time Live Sync
ALTER PUBLICATION supabase_realtime ADD TABLE public.tugas_kuliah;
